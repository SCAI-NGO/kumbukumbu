# litmus tests that fail by design (WebDAV design §3.4, §3.7, §7), as suite/number.name; tools/litmus/run.sh fails
# on any other failure and on any of these that passes. The numbers are litmus 0.13's.

# PROPPATCH answers 207 with every property set and stores nothing (§3.4): Windows and Finder set times and their own
# properties, and refusing breaks their copies. So a PROPFIND after it finds no dead property.
props/7.propget    # dead properties are not stored
props/10.propget   # dead properties are not stored (after a MOVE)
props/12.propget   # dead properties are not stored (after deleting some)
props/14.propget   # dead properties are not stored (after replacing some)
props/16.propget   # dead properties are not stored (one without a namespace)
props/18.propget   # dead properties are not stored (a value outside the BMP)
props/20.propget   # dead properties are not stored (removed and set in one PROPPATCH)
props/27.propget   # dead properties are not stored (many namespaces)

# Fake locking (§3.7): LOCK hands out a token and UNLOCK takes it back, but a lock is never enforced, so another
# client may write, delete and lock what is locked.
locks/9.notowner_modify    # locks are not enforced: DELETE of an exclusively locked file succeeds
locks/10.notowner_lock     # locks are not enforced: a second exclusive LOCK succeeds
locks/12.notowner_modify   # locks are not enforced (after the owner wrote)
locks/13.notowner_lock     # locks are not enforced (after the owner wrote)
locks/24.notowner_modify   # locks are not enforced: DELETE of a file with a shared lock succeeds
locks/25.notowner_lock     # locks are not enforced: an exclusive LOCK over a shared one succeeds
locks/28.notowner_modify   # locks are not enforced (two shared locks)
locks/29.notowner_lock     # locks are not enforced (two shared locks)
locks/34.notowner_modify   # locks are not enforced: DELETE in a locked folder succeeds

# If headers with lock tokens are accepted and ignored (§3.7), so a condition that names a wrong or no longer valid
# token doesn't fail the request.
locks/16.fail_cond_put           # If headers are ignored: a PUT with an unknown lock token succeeds
locks/18.cond_put_corrupt_token  # If headers are ignored: a PUT with a malformed lock token succeeds
locks/20.fail_complex_cond_put   # If headers are ignored: a PUT whose If lists only false conditions succeeds
locks/22.fail_cond_put_unlocked  # If headers are ignored: a PUT with the token of an ended lock succeeds
